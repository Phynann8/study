មេរៀនទី ១: Exception Handling (ការដោះស្រាយបញ្ហា)

នៅក្នុងជីវិតពិត រឿងដែលមិនរំពឹងទុកតែងតែកើតឡើង ដូចជាការកំពប់ទឹក ឬបែកកែវ។ នៅក្នុងការសរសេរកូដក៏ដូចគ្នា យើងហៅបញ្ហាទាំងនេះថា Exceptions។

យើងប្រើ try, catch, និង finally ដើម្បីដោះស្រាយបញ្ហាទាំងនេះដោយមិនឲ្យកម្មវិធីគាំង។

របៀបប្រើប្រាស់ (Syntax)


try 
{
    // កូដដែលអាចមានបញ្ហា
    int[] myNumbers = {1, 2, 3};
    Console.WriteLine(myNumbers[10]); // កំហុស! ព្រោះគ្មានលេខទី 10
}
catch (Exception e)
{
    // ដោះស្រាយបញ្ហា
    Console.WriteLine("មានបញ្ហាកើតឡើង: " + e.Message);
}
finally 
{
    // ធ្វើជានិច្ច ទោះមានរឿងអ្វីក៏ដោយ
    Console.WriteLine("ចប់ការងារ។");
}

ការពន្យល់:

  • try: សាកល្បងធ្វើអ្វីមួយ។
  • catch: ចាប់យកបញ្ហាប្រសិនបើវាកើតឡើង។
  • finally: អនុវត្តកូដចុងក្រោយជានិច្ច (ដូចជាការបិទទ្វារផ្ទះពេលចេញក្រៅ)។

សាកល្បងគិតមើល!
