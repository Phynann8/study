មេរៀនទី ២: File I/O (ការអាន និងសរសេរឯកសារ)

កម្មវិធីភាគច្រើនត្រូវការរក្សាទុកទិន្នន័យ។ File I/O (Input/Output) គឺជាការសរសេរទិន្នន័យចូលទៅក្នុងឯកសារ (Write) និងអានវាមកវិញ (Read)។

យើងប្រើ class File នៅក្នុង System.IO។

កូដគំរូ (Example)


using System.IO;

// សរសេរអក្សរទៅក្នុងឯកសារ
string writeText = "សួស្តី! នេះគឺជាការសាកល្បងសរសេរឯកសារ។";
File.WriteAllText("filename.txt", writeText);

// អានអក្សរពីឯកសារ
string readText = File.ReadAllText("filename.txt");
Console.WriteLine(readText);

ចំណុចសំខាន់ៗ:

  • WriteAllText: បង្កើតឯកសារថ្មី ឬសរសេរជាន់លើឯកសារចាស់។
  • AppendAllText: សរសេរបន្ថែមពីក្រោម ដោយមិនលុបទិន្នន័យចាស់។
  • ReadAllText: អានអក្សរទាំងអស់នៅក្នុងឯកសារ។
