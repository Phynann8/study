មេរៀនទី ៣: តក្កវិជ្ជា (Logic) - ការតថ្លៃនៅផ្សារ

នៅពេលអ្នកទៅផ្សារ អ្នកតែងតែមានការសម្រេចចិត្ត៖ "ប្រសិនបើ (If) តម្លៃថោក ខ្ញុំនឹងទិញ ប៉ុន្តែប្រសិនបើ (Else) ថ្លៃពេក ខ្ញុំនឹងមិនទិញទេ"។

នៅក្នុងកូដ យើងប្រើ if និង else ដើម្បីឱ្យកម្មវិធីចេះសម្រេចចិត្តតាមលក្ខខណ្ឌដែលយើងកំណត់។

របៀបប្រើ If...Else (Syntax)


int price = 5000;

if (price <= 4000) 
{
    Console.WriteLine("ទិញត្រីនេះ!");
}
else 
{
    Console.WriteLine("ថ្លៃពេកហើយ សុំចុះថ្លៃបន្តិចបានទេ?");
}

ការពន្យល់:

  • price <= 4000: គឺជាលក្ខខណ្ឌ (ដូចជាការកំណត់តម្លៃក្នុងចិត្តរបស់អ្នក)។
  • if: ប្រសិនបើលក្ខខណ្ឌខាងលើ ពិត (True) វានឹងធ្វើការងារក្នុងចន្លោះ { } ដំបូង។
  • else: ប្រសិនបើលក្ខខណ្ឌខាងលើ មិនពិត (False) វានឹងធ្វើការងារក្នុងចន្លោះ { } បន្ទាប់។

សាកល្បងតថ្លៃ!
