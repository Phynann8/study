មេរៀនទី ២.១: Classes & Objects - ប្លង់មេនៃហាង

នៅក្នុងមេរៀនមុន យើងបានរៀនអំពី "ប្រអប់" (Variables) ។ ឥឡូវនេះ ស្រមៃថាអ្នកចង់បើកហាងលក់អីវ៉ាន់។

តើអ្នកគួរធ្វើដូចម្តេចដើម្បីគ្រប់គ្រងអីវ៉ាន់រាប់ពាន់មុខ? អ្នកត្រូវការ "ប្លង់មេ (Blueprint)" ។

១. Class គឺជាអ្វី? (ប្លង់មេ)

Class គឺដូចជាប្លង់មេនៃផ្ទះ ឬរូបមន្តធ្វើម្ហូប។ វាមិនមែនជាផ្ទះពិតប្រាកដទេ ប៉ុន្តែវាប្រាប់យើងថាផ្ទះនោះមានអ្វីខ្លះ (បង្អួចប៉ុន្មាន ទ្វារប៉ុន្មាន)។


class Item 
{
    public string Name;
    public int Price;
}

២. Object គឺជាអ្វី? (របស់ពិត)

Object គឺជាអ្វីដែលយើងបង្កើតចេញពី Class។ បើ Class ជាប្លង់ផ្ទះ Object គឺជា "ផ្ទះពិតប្រាកដ" ដែលយើងអាចរស់នៅបាន។ 


Item coca = new Item();
coca.Name = "Coca Cola";
coca.Price = 2500;

សាកល្បងគិតមើល!
